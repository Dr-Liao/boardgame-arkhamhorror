```js
// 08131 腿伤（Leg Injury）
// 卡面：显现 - 将腿伤放置入场，放到你的威胁区域。
//        腿伤可被治愈，如同是你身上的 1 点伤害（如果它被治愈，将其弃掉）。
//        强制 - 在你执行移动、撤退或躲避行动后：本回合剩余时间内，你不能再进行任何这些类型的行动。
//
// 一张卡一个文件：这张卡的行为只在这里维护。
// 不要往 card-abilities.js 或任何批次文件里追加条目；新增卡用 tools/scaffold-card.mjs <code> --entry。
//
// 读点：弱点模板 kind:'weakness'（type:'threat' + clues:1）、actionResolvedAtLocation 窗口、
//       api.addEffect（capability:move / resign / evade，duration:'endOfTurn'）。
// approximation:
//   ① 「如同 1 点伤害被治愈」：无 threatDamageSink，用 free + discardSelf 近似。
//   ② 「执行移动/撤退/躲避行动后」：窗口不带行动类型，近似为任意行动后锁住印刷三类行动至回合结束。
const code = '08131'

export default {
  code,
  note: 'approximation: 伤害侧治愈用「弃掉自身」近似（无 threatDamageSink）；动作后锁近似为任意行动后锁住印刷行动类型至回合结束。',
  abilities: [
    {
      kind: 'weakness',
      label: '显现：放入威胁区域（1 点「伤害」计数器）',
      weakness: { type: 'threat', clues: 1 },
    },
    {
      kind: 'free',
      label: '近似-可治愈：弃掉腿伤（视为其 1 点伤害被治愈）',
      cost: { action: false, discardSelf: true },
    },
    {
      kind: 'forced',
      trigger: 'actionResolvedAtLocation',
      label: '执行行动后：本回合不能移动、撤退或躲避',
      criteria: [({ state, inv }) => !(state.effects || []).some(effect => effect.sourceCode === code
        && effect.duration === 'endOfTurn' && effect.controllerSeat === inv.seat)],
      resolve: ({ inv, api }) => {
        api.addEffect({
          sourceCode: code,
          controllerSeat: inv.seat,
          duration: 'endOfTurn',
          label: '腿伤：本回合不能移动、撤退或躲避',
          modifiers: [
            { target: 'capability:move' },
            { target: 'capability:resign' },
            { target: 'capability:evade' },
          ],
        })
        api.info(`🩹 「腿伤」：${inv.name} 本回合不能再移动、撤退或躲避。`)
      },
    },
  ],
  modifiers: [],
}

```
