## 条目源码
```js
// 08028 追猎行动
// 卡面：快速。在神话阶段期间你将抽取遭遇卡牌时打出。
//       改为从遭遇牌库中查找一个敌人，将其生成并与你交战(而非在本来的位置生成它)，
//       将追猎行动附属到该敌人上，并混洗遭遇牌库。
//       [reaction] 在你击败被附属的敌人时：获得 3 资源。
//
// 一张卡一个文件：这张卡的行为只在这里维护。
// 不要往 card-abilities.js 或任何批次文件里追加条目；新增卡用 tools/scaffold-card.mjs <code> --entry。
//
// approximation：
//   · 遭遇牌库「查找」用只读 state.encounterDeck 枚举敌人 code + api.spawnEncounterEnemy（生成/交战/混洗
//     都在该入口内完成）；
//   · 没有「结算后的事件卡附到敌人」原语（请求 m2-1-08028.json），本实现不生成实体附属卡：用持续效果
//     记录的 sourceInstanceId 指向被追猎的敌人来标记归属（敌人实体上没有 attached 卡，存档/投影里不可见）；
//   · 击败奖励用该持续效果上的 defeatEnemy 反应（effect DSL gainResource 3）表达——效果有效期内任意
//     敌人被击败都可触发一次（近似为「被附属敌人被击败」；duration endOfRound + oncePerGame 限制滥触发）。
const isEnemy = (api, code) => api.cardType(code) === 'enemy'

export default {
  code: '08028',
  abilities: [
    {
      kind: 'played',
      fast: true,
      trigger: 'mythosEncounterStep',
      label: '神话阶段抽遭遇前：从遭遇牌库查找一个敌人生成并与你交战，附属本卡；击败被附属敌人获得 3 资源',
      resolve: ({ state, inv, api }) => {
        const candidates = [...new Set((state.encounterDeck || []).filter(code => isEnemy(api, code)))]
        if (!candidates.length) {
          api.info('追猎行动：遭遇牌库中没有敌人。')
          return
        }
        const chosen = candidates.length === 1 ? candidates[0] : api.choice({
          kind: 'encounter', source: '08028', seat: inv.seat,
          prompt: '追猎行动：从遭遇牌库中查找一个敌人。',
          options: candidates.map(code => ({ id: code, cardCode: code, label: api.cardName(code) })),
        })
        const code = candidates.includes(chosen) ? chosen : candidates[0]
        const spawned = api.spawnEncounterEnemy(inv.seat, code)
        if (!spawned) {
          api.info('追猎行动：无法生成该敌人。')
          return
        }
        api.info(`追猎行动：「${api.cardName(code)}」在 ${inv.name} 的地点生成并交战；遭遇牌库已混洗。`)
        api.addEffect({
          duration: 'endOfRound',
          sourceCode: '08028',
          sourceInstanceId: spawned.id,
          label: '追猎行动：击败被追猎的敌人获得 3 资源',
          modifiers: [],
          abilities: [{
            kind: 'reaction',
            trigger: 'defeatEnemy',
            label: '击败被追猎的敌人：获得 3 资源',
            limit: 'oncePerGame',
            effect: { type: 'gainResource', amount: 3 },
          }],
        })
      },
    },
  ],
  modifiers: [],
}

```
