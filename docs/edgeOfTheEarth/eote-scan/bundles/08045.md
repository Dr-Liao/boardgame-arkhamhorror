## 条目源码
```js
// 08045 亵渎预言
// 卡面：你不在坐标点时，你得到 +1[intellect] 和 +1[agility]，并且你可以忽略趁乱攻击。
//       [reaction] 在亵渎预言入场后：选择一个已揭示地点。该地点是“坐标点”，直到亵渎预言离场。
//       [action]：移动任意一位调查员至坐标点。
//
// 一张卡一个文件：这张卡的行为只在这里维护。
// 不要往 card-abilities.js 或任何批次文件里追加条目；新增卡用 tools/scaffold-card.mjs <code> --entry。
//
// 口径：「坐标点」不新建单卡状态字段，而是用已登记的地点特性原语 api.addLocationTrait 授予
// [[Coordinate]] 特性（来源记 08045），离场时 api.removeLocationTrait 撤销。修正与忽略趁乱攻击的
// condition 都读同一个地点特性来源。行动选择任意存活调查员（不限于同地点）移动到坐标点。
//
// approximation：没有通用的「地点标记」原语，这里借用地点特性记账；「坐标点」会以被授予的地点特性
// 出现在地点特性投影里（投射为 Coordinate 特性而不是独立的坐标点标记）。卡面其余效果（+1/+1、
// 忽略趁乱攻击、移动到坐标点、离场撤销）均按现有读点正确接线。
const COORDINATE_TRAIT = 'Coordinate'
const SOURCE = '08045'

const coordinateLocationId = state => {
  const location = (state?.locations || []).find(item =>
    (item.traitGrants?.[COORDINATE_TRAIT] || []).includes(SOURCE))
  return location?.id || null
}
const awayFromCoordinate = ({ state, inv }) => {
  const id = coordinateLocationId(state)
  return Boolean(id) && inv?.location !== id
}

export default {
  code: '08045',
  modifiers: [
    { target: 'skill:intellect', amount: 1, condition: { when: awayFromCoordinate } },
    { target: 'skill:agility', amount: 1, condition: { when: awayFromCoordinate } },
    { target: 'ignore:opportunityAttack', amount: 1, condition: { when: awayFromCoordinate } },
  ],
  abilities: [
    {
      kind: 'entersPlay',
      label: '入场：选择一个已揭示地点作为“坐标点”',
      resolve: ({ state, inv, source, api }) => {
        if (!source) return
        const candidates = (state.locations || []).filter(location => location.revealed === true)
        if (!candidates.length) {
          api.info('亵渎预言：没有已揭示地点可选为坐标点。')
          return
        }
        const answer = candidates.length === 1 ? candidates[0].id : api.choice({
          kind: 'location', source: SOURCE, seat: inv.seat,
          prompt: '亵渎预言：选择一个已揭示地点作为“坐标点”。',
          options: candidates.map(location => ({ id: location.id, label: api.locName(location.id) })),
        })
        const target = candidates.find(location => location.id === answer) || candidates[0]
        if (!api.addLocationTrait(target, COORDINATE_TRAIT, { source: SOURCE })) return
        api.info(`亵渎预言：「${api.locName(target.id)}」成为坐标点，直到亵渎预言离场。`)
      },
    },
    {
      kind: 'leavesPlay',
      label: '离场：撤销“坐标点”',
      resolve: ({ state, api }) => {
        for (const location of state.locations || []) {
          if ((location.traitGrants?.[COORDINATE_TRAIT] || []).includes(SOURCE)) {
            api.removeLocationTrait(location, COORDINATE_TRAIT, { source: SOURCE })
          }
        }
      },
    },
    {
      kind: 'action',
      label: '移动任意一位调查员至坐标点',
      cost: { action: true },
      resolve: ({ state, inv, api }) => {
        const coordinateId = coordinateLocationId(state)
        if (!coordinateId) {
          api.info('亵渎预言：还没有坐标点。')
          return
        }
        const candidates = (state.investigators || []).filter(person => !person.defeated && !person.resigned)
        const answer = candidates.length === 1 ? String(candidates[0].seat) : api.choice({
          kind: 'investigator', source: SOURCE, seat: inv.seat,
          prompt: '亵渎预言：选择要移动到坐标点的调查员。',
          options: candidates.map(person => ({ id: String(person.seat), label: person.name })),
        })
        const target = candidates.find(person => String(person.seat) === String(answer)) || candidates[0]
        api.moveInvestigator(target.seat, coordinateId)
        api.info(`亵渎预言：将 ${target.name} 移动到坐标点「${api.locName(coordinateId)}」。`)
      },
    },
  ],
}

```
