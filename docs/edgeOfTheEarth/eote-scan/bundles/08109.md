## 条目源码
```js
// 08109 身形虚化（Blur，0–1 级）
// 卡面：使用(3充能)。
// [action] 如果身形虚化还有剩余充能：躲避。你在本次躲避尝试中可以不使用[agility]，改为使用[willpower]，
//        并且得到 +1 技能值。如果你成功，花费 1 充能且本回合可以执行 1 次额外行动。
//        如果你成功且等于难度，受到 1 点伤害。
//
// 一张卡一个文件：这张卡的行为只在这里维护。
// 不要往 card-abilities.js 或任何批次文件里追加条目；新增卡用 tools/scaffold-card.mjs <code> --entry。
//
// 读点：
//   · 躲避走 api.evade（支持 skill/bonus/assetId，返回真实检定结果）；
//   · 花充能走 api.spendUses，额外行动走 api.gainAction（P0–P5 既有原语）；
//   · 「成功且等于难度」读检定结果的 value/difficulty，不写私有回合字段。
const code = '08109'

export default {
  code,
  uses: 3,
  usesType: 'charges',
  abilities: [
    {
      kind: 'action',
      label: '躲避：可改用意志、+1 技能；成功花 1 充能获得额外行动',
      countsAs: 'evade',
      cost: { action: true },
      // 「如果身形虚化还有剩余充能」：没有充能时能力不可用。
      criteria: [({ source }) => (source?.uses || 0) > 0],
      target: { kind: 'enemy', query: ({ inv, entity }) => entity.engagedSeat === inv.seat },
      resolve: ({ inv, source, target, api }) => {
        const enemy = target?.entity
        if (!enemy) return
        const skill = inv.willpower === inv.agility
          ? 'agility'
          : api.choice({
            kind: 'effect', source: code, seat: inv.seat,
            prompt: '身形虚化：本次躲避尝试使用哪个技能？',
            options: [{ id: 'agility', label: '使用敏捷' }, { id: 'willpower', label: '改用意志' }],
          })
        const result = api.evade(inv.seat, {
          enemyId: enemy.id,
          skill: skill === 'willpower' ? 'willpower' : 'agility',
          bonus: 1,
          assetId: source?.id,
          sourceCode: code,
        })
        if (!result?.success) return
        api.spendUses(source, 1)
        api.gainAction(inv.seat, 1)
        api.info(`🌀 「身形虚化」：${inv.name} 躲避成功，花费 1 充能并获得 1 个额外行动。`)
        if (result.value - result.difficulty === 0) {
          api.damage(inv.seat, 1)
          api.info(`🌀 「身形虚化」：成功且等于难度，${inv.name} 受到 1 点伤害。`)
        }
      },
    },
  ],
  modifiers: [],
}

```
