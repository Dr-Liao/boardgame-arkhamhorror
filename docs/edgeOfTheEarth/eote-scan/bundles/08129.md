// 08129 请求支援（Call for Backup）
// 卡面：以任意顺序依次结算，如果你控制……
// - [rogue] 卡牌，你可以移动到一个连接地点。
// - [guardian] 卡牌，对你所在地点的一个敌人造成 1 点伤害。
// - [seeker] 卡牌，发现你所在地点的 1 个线索。
// - [mystic] 卡牌，治愈任意卡牌的 1 点恐惧。
// - [survivor] 卡牌，治愈任意卡牌的 1 点伤害。
//
// 一张卡一个文件：这张卡的行为只在这里维护。
// 不要往 card-abilities.js 或任何批次文件里追加条目；新增卡用 tools/scaffold-card.mjs <code> --entry。
//
// 读点：
//   · 职阶判定用 api.cardFactions（在场支援卡面数据；不按卡号分支）；
//   · 位移 api.moveInvestigator、伤害 api.damageEnemy、线索 api.discoverClue、
//     治疗 api.healDamage / api.healHorror / api.healAsset（支援恐惧走统一入口）。
const code = '08129'
const BULLETS = [
  { id: 'rogue', faction: 'rogue', label: '[rogue] 卡牌：移动到一个连接地点' },
  { id: 'guardian', faction: 'guardian', label: '[guardian] 卡牌：对所在地点一个敌人造成 1 点伤害' },
  { id: 'seeker', faction: 'seeker', label: '[seeker] 卡牌：发现所在地点的 1 个线索' },
  { id: 'mystic', faction: 'mystic', label: '[mystic] 卡牌：治愈任意卡牌的 1 点恐惧' },
  { id: 'survivor', faction: 'survivor', label: '[survivor] 卡牌：治愈任意卡牌的 1 点伤害' },
]

function healTargets(state, inv, kind) {
  const field = kind === 'damage' ? 'damage' : 'horror'
  const targets = []
  for (const owner of state.investigators) {
    if (owner.defeated || owner.resigned) continue
    if ((owner[field] || 0) > 0) {
      targets.push({ id: `investigator:${owner.seat}`, label: `${owner.name}（调查员）`, owner, asset: null })
    }
    for (const assetCard of owner.assets || []) {
      if ((assetCard[field] || 0) > 0) {
        targets.push({ id: `asset:${assetCard.id}`, label: `${assetCard.name || assetCard.code}（支援）`, owner, asset: assetCard })
      }
    }
  }
  return targets
}

export default {
  code,
  abilities: [
    {
      kind: 'played',
      label: '按任意顺序结算你控制职阶对应的援助效果',
      resolve: ({ state, inv, api }) => {
        const controlled = new Set((inv.assets || []).flatMap(card => api.cardFactions(card.code)))
        const remaining = BULLETS.filter(bullet => controlled.has(bullet.faction))
        if (!remaining.length) {
          api.info('「请求支援」：你没有控制任何对应职阶的卡牌，无事发生。')
          return
        }
        while (remaining.length) {
          const answer = api.choice({
            kind: 'effect', source: code, seat: inv.seat,
            prompt: '「请求支援」：按任意顺序结算一项（或结束）。',
            options: [{ id: 'done', label: '结束结算' },
              ...remaining.map(bullet => ({ id: bullet.id, label: bullet.label }))],
          })
          if (!answer || answer === 'done') break
          const index = remaining.findIndex(bullet => bullet.id === answer)
          if (index < 0) break
          const [bullet] = remaining.splice(index, 1)
          if (bullet.id === 'rogue') {
            const here = state.locations.find(location => location.id === inv.location)
            const destinations = (here?.connections || [])
              .map(id => state.locations.find(location => location.id === id))
              .filter(Boolean)
            if (!destinations.length) continue
            const destination = destinations.length === 1 ? destinations[0].id : api.choice({
              kind: 'move', source: code, seat: inv.seat,
              prompt: '「请求支援」：选择连接地点。',
              options: destinations.map(location => ({ id: location.id, label: location.name || location.id })),
            })
            if (destination) api.moveInvestigator(inv.seat, destination)
          } else if (bullet.id === 'guardian') {
            const enemies = state.enemies.filter(enemy => !enemy.defeated && enemy.location === inv.location)
            if (!enemies.length) continue
            const enemyId = enemies.length === 1 ? enemies[0].id : api.choice({
              kind: 'target', source: code, seat: inv.seat,
              prompt: '「请求支援」：选择要造成 1 点伤害的敌人。',
              options: enemies.map(enemy => ({ id: enemy.id, label: enemy.name })),
            })
            const enemy = enemies.find(item => item.id === enemyId)
            if (enemy) api.damageEnemy(enemy, 1, inv.seat, { code })
          } else if (bullet.id === 'seeker') {
            const location = state.locations.find(item => item.id === inv.location)
            if ((location?.clues || 0) > 0) api.discoverClue(inv.seat, 1)
          } else {
            const kind = bullet.id === 'mystic' ? 'horror' : 'damage'
            const targets = healTargets(state, inv, kind)
            if (!targets.length) continue
            const chosen = targets.length === 1 ? targets[0].id : api.choice({
              kind: 'target', source: code, seat: inv.seat,
              prompt: `「请求支援」：选择要治愈 1 点${kind === 'horror' ? '恐惧' : '伤害'}的卡牌。`,
              options: targets.map(target => ({ id: target.id, label: target.label })),
            })
            const target = targets.find(item => item.id === chosen)
            if (!target) continue
            if (target.asset) api.healAsset(target.asset, { [kind]: 1, owner: target.owner })
            else if (kind === 'horror') api.healHorror(target.owner.seat, 1)
            else api.healDamage(target.owner.seat, 1)
          }
        }
      },
    },
  ],
  modifiers: [],
}
