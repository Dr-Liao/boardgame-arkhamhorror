```js
// 08074 应急装备（Jury-Rig）· 幸存者事件 · 升级 · 0 经验。
//
// 卡面：使用(3 耐久)。附属到你所在地点的一位调查员控制的 [[道具]] 支援上。
//       [free] 被附属支援的技能检定期间，花费 1 耐久：你本次检定得到 +2 技能值。
//
// 一卡一文件：这张卡的行为只在这里维护。
// approximation: 引擎的 attachTo 事件通道不会按条目 uses 初始化附属实例，所以这里在 played
// resolve 里手动 api.moveCard 到 attached 并用 api.setAssetUses 设置 3 耐久；「被附属支援的
// 技能检定期间」用 state.skillTest.source.assetId 或 state._resolvingAssetId 与宿主 id 比对
// （仅对通过支援能力发起的检定可靠，见报告）。
import { ALL_CARDS } from '../../cards/cards-data.js'

const isItemAsset = code => {
  const def = ALL_CARDS[code]
  return Boolean(def) && def.type === 'asset' && (def.traits || []).some(trait => String(trait).toLowerCase() === 'item')
}

export default {
  code: '08074',
  uses: 3,
  usesType: 'durability',
  abilities: [
    {
      kind: 'played',
      label: '附属到你所在地点一位调查员控制的[[道具]]支援（3 耐久）',
      resolve: ({ state, inv, api }) => {
        const hosts = state.investigators
          .filter(owner => !owner.defeated && !owner.resigned && owner.location === inv.location)
          .flatMap(owner => (owner.assets || []).map(asset => ({ owner, asset })))
          .filter(({ asset }) => isItemAsset(asset.code))
        if (!hosts.length) {
          api.info('「应急装备」：同地点没有可以附属的[[道具]]支援。')
          return
        }
        const answer = hosts.length === 1 ? '0'
          : api.choice({ kind: 'target', source: '08074', seat: inv.seat,
            prompt: '选择要附属的[[道具]]支援。',
            options: hosts.map(({ owner, asset }, index) => ({ id: String(index), label: `${api.cardName(asset.code)}（${owner.name}）` })) })
        const target = hosts[Number(answer)] || hosts[0]
        if (!target) return
        const moved = api.moveCard({ code: '08074', from: 'hand', to: 'attached', attachTo: target.asset.id })
        if (!moved) {
          api.info('「应急装备」：无法附属到该支援。')
          return
        }
        const attachment = (target.asset.attachments || []).at(-1)
        if (attachment) api.setAssetUses(attachment, 3)
        api.info(`🧰 「应急装备」：附属到「${api.cardName(target.asset.code)}」（3 耐久）。`)
      },
    },
    {
      kind: 'free',
      cost: { uses: 1 },
      label: '被附属支援的技能检定期间：花 1 耐久，本次检定 +2 技能值',
      criteria: [({ state, source }) => {
        const hostId = source?.hostAssetId
        if (!hostId || !state.skillTest) return false
        return state.skillTest.source?.assetId === hostId || state._resolvingAssetId === hostId
      }],
      resolve: ({ inv, api }) => {
        api.addPendingTestBonus(2)
        api.info(`🧰 「应急装备」：${inv.name} 本次技能检定 +2 技能值。`)
      },
    },
  ],
  modifiers: [],
}

```
