## 条目源码
```js
// 08102 保护魂灵
// 卡面：快速。你所在地点的一个[[盟友]]支援被击败或因卡牌效果弃掉后打出。
//       将该支援返回其拥有者手中，或其拥有者抽取3张卡牌。
//
// 读点：assetLeavesPlay 窗口（含 reason）；返回手牌 api.moveCard({from:'discard',to:'hand',ownerSeat})，
// 或 api.drawCards(ownerSeat,3)。
//
// 一张卡一个文件：这张卡的行为只在这里维护。

export default {
  code: '08102',
  abilities: [
    { kind: 'played', fast: true, trigger: 'assetLeavesPlay', label: '盟友支援离场后：返回其拥有者手中，或让拥有者抽 3 张',
      criteria: ({ inv, context }) => Boolean(context?.asset) && context?.seat === inv.seat
        && context?.asset?.location === inv.location
        && ((context?.asset?.traits || []).some(trait => String(trait).toLowerCase() === 'ally')),
      resolve: ({ inv, source, context, api }) => {
        const asset = context?.asset
        if (!asset) return
        const owner = Number.isSafeInteger(asset.ownerSeat) ? asset.ownerSeat : inv.seat
        const picked = api.choice({ kind: 'effect', source: '08102', seat: inv.seat,
          prompt: '选择保护魂灵的效果。', options: [
            { id: 'return', label: '将该支援返回其拥有者手中' },
            { id: 'draw', label: `让拥有者抽取 3 张卡牌` },
          ] })
        if (picked === 'return') api.moveCard({ code: asset.code, from: 'discard', to: 'hand', ownerSeat: owner })
        else api.drawCards(owner, 3)
      } },
  ],
  modifiers: [],
}

```
