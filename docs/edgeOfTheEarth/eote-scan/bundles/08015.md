## 条目源码
```js
// 08015 命运重担（陈丽丽专属弱点）
// 卡面：显现 - 你必须选择一项：
//   · 将你控制的一张演武翻至 Broken 面。它本轮不能翻回 Unbroken 面。
//   · 承受 1 点伤害和 1 点恐惧。
//
// 一张卡一个文件：这张卡的行为只在这里维护。
// 不要往 card-abilities.js 或任何批次文件里追加条目；新增卡用 tools/scaffold-card.mjs <code> --entry。

// 「本轮不能翻回」用一条 endOfRound 持续效果当锁标记（纯 JSON；Broken 面的轮末反应读同一份记录）。
const LOCK_LABEL = '命运重担：本轮不能翻回'

const flipToBroken = (api, asset, unbroken) => {
  const broken = `${unbroken.slice(0, -1)}b`
  api.flipAsset(asset, broken)
  if (api.mechanics?.disciplines?.faceCode?.(unbroken) != null) api.mechanics.disciplines.flip(unbroken, { to: 'broken' })
}

export default {
  code: '08015',
  abilities: [
    {
      kind: 'weakness',
      label: '命运重担',
      weakness: { type: 'burdenOfDestiny' },
      // 自定义结算接管整段显现：二选一（翻演武并锁定本轮 / 承受 1 伤害 1 恐惧）。
      // 读点：src/engine/weakness.js 的 resolveWeaknessBody 先找 descriptor.resolve（顶层字段）。
      resolve: ({ inv, api }) => {
        const candidates = (inv.assets || []).filter(asset => /^0801[1-4]a$/.test(String(asset.code || '')))
        const options = []
        if (candidates.length) options.push({ id: 'flip', label: '将一张演武翻至 Broken 面；它本轮不能翻回' })
        options.push({ id: 'harm', label: '承受 1 点伤害和 1 点恐惧' })
        const choice = options.length === 1 ? 'harm' : api.choice({
          kind: 'effect', source: '08015', seat: inv.seat,
          prompt: '命运重担：选择一项。', options,
        })
        if (choice === 'flip') {
          let asset = candidates[0]
          if (candidates.length > 1) {
            const index = Number(api.choice({
              kind: 'effect', source: '08015', seat: inv.seat,
              prompt: '选择要翻至 Broken 面的演武。',
              options: candidates.map((candidate, i) => ({ id: String(i), label: api.cardName(candidate.code) })),
            }))
            asset = candidates[index] || candidates[0]
          }
          flipToBroken(api, asset, String(asset.code))
          api.addEffect({
            duration: 'endOfRound', sourceCode: '08015', sourceInstanceId: asset.id,
            controllerSeat: inv.seat, label: LOCK_LABEL, modifiers: [],
          })
          return
        }
        api.damage(inv.seat, 1)
        api.horror(inv.seat, 1)
      },
    },
  ],
  modifiers: [],
}

```
