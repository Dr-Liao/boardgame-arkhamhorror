## 条目源码
```js
// 08100 远古海图（4 级）
// 卡面：使用(3秘密)。
//       [free]在你回合期间，横置远古海图并花费1秘密：以任意顺序选择并执行以下两个行动(移动、躲避或调查)。
//
// 读点同 08098：移动/躲避/调查都走不消耗行动的免费执行入口；本卡执行两轮选择。
//
// 一张卡一个文件：这张卡的行为只在这里维护。

const performAction = (inv, state, api) => {
  const engaged = state.enemies.filter(enemy => inv.engaged.includes(enemy.id) && !enemy.exhausted && !enemy.defeated)
  const options = [{ id: 'move', label: '移动' }, { id: 'investigate', label: '调查' }]
  if (engaged.length) options.push({ id: 'evade', label: '躲避' })
  const picked = api.choice({ kind: 'effect', source: '08100', seat: inv.seat, prompt: '选择要执行的行动。', options })
  if (picked === 'move') {
    const destinations = api.moveDestinations(inv.seat)
    if (!destinations.length) return
    const destination = destinations.length === 1 ? destinations[0] : api.choice({ kind: 'location', source: '08100', seat: inv.seat,
      prompt: '选择移动目的地。', options: destinations.map(id => ({ id, label: api.locName(id) })) })
    api.moveInvestigator(inv.seat, destination)
    return
  }
  if (picked === 'evade') {
    const enemyId = engaged.length === 1 ? engaged[0].id : api.choice({ kind: 'target', source: '08100', seat: inv.seat,
      prompt: '选择要躲避的敌人。', options: engaged.map(enemy => ({ id: enemy.id, label: enemy.name })) })
    api.evade(inv.seat, { enemyId })
    return
  }
  api.investigate(inv.seat)
}

export default {
  code: '08100',
  uses: 3,
  usesType: 'secrets',
  abilities: [
    { kind: 'free', label: '横置并花费 1 秘密：执行两个移动/躲避/调查行动', cost: { action: false, exhaust: true, uses: 1 },
      resolve: ({ inv, state, api }) => {
        performAction(inv, state, api)
        performAction(inv, state, api)
      } },
  ],
  modifiers: [],
}

```
