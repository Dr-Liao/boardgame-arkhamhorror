## 条目源码
```js
// 08038 医学教科书
// 卡面：[action]：选择你所在地点的一位调查员并检定[intellect](2)。如果你成功，治愈该调查员
//       1 点伤害（如果你成功并超过难度至少 2 点，改为治愈 2 点伤害）。如果你失败，你必须横置
//       医学教科书或对该调查员造成 1 点伤害。
//
// 一张卡一个文件：这张卡的行为只在这里维护。
// 不要往 card-abilities.js 或任何批次文件里追加条目；新增卡用 tools/scaffold-card.mjs <code> --entry。
//
// 口径：选择同地点调查员用 api.investigatorsAt(inv.location)；检定用 api.startTest（intellect 2），
// 成功差值 >=2 时治愈 2；失败时必须二选一（横置本卡 / 对该调查员 1 点伤害）。
export default {
  code: '08038',
  abilities: [
    {
      kind: 'action',
      label: '选择同地点一位调查员：检定[智力](2)，成功治愈 1（超过 2 点则 2）；失败横置或对其造成 1 伤害',
      cost: { action: true },
      resolve: ({ inv, source, api }) => {
        const candidates = api.investigatorsAt(inv.location)
        if (!candidates.length) {
          api.info('医学教科书：这里没有调查员。')
          return
        }
        const answer = candidates.length === 1 ? String(candidates[0].seat) : api.choice({
          kind: 'investigator', source: source.id, seat: inv.seat,
          prompt: '医学教科书：选择你所在地点的一位调查员。',
          options: candidates.map(other => ({ id: String(other.seat), label: other.name })),
        })
        const targetSeat = Number(answer)
        api.startTest({
          seat: inv.seat, skill: 'intellect', difficulty: 2, source: { label: '医学教科书', code: '08038' },
          resolve: result => {
            if (result?.success) {
              const amount = (result.value - result.difficulty) >= 2 ? 2 : 1
              api.healDamage(targetSeat, amount)
              api.info(`医学教科书：治愈 ${api.investigatorsAt(inv.location).find(other => other.seat === targetSeat)?.name || '调查员'} ${amount} 点伤害。`)
              return
            }
            const answerFail = api.choice({
              kind: 'effect', source: source.id, seat: inv.seat,
              prompt: '医学教科书：检定失败 —— 横置医学教科书，或对该调查员造成 1 点伤害。',
              options: [
                { id: 'exhaust', label: '横置医学教科书' },
                { id: 'damage', label: `对「${api.investigatorsAt(inv.location).find(other => other.seat === targetSeat)?.name || '调查员'}」造成 1 点伤害` },
              ],
            })
            if (answerFail === 'damage') api.damage(targetSeat, 1)
            else api.exhaust(source)
          },
        })
      },
    },
  ],
  modifiers: [],
}

```
