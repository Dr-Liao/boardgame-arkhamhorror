## 条目源码
```js
// 08005 伊波恩之书（Livre d'Eibon）· 诺曼专属支援。
//
// 卡面：
//   [fast] 横置伊波恩之书：用你一张手牌交换你牌库顶的卡牌。
//   [fast] 横置伊波恩之书：将你牌库顶的卡牌投入到你所在地点一位调查员执行的符合条件的技能检定中。
//
// 引擎读点：牌库顶状态由 P2 `eote-deck-top.js` 维护（scenarioFlags.mechanics.eote.deckTop）；
// 换牌用 api.moveCard + api.deckTop.clear/reveal 组合；投入牌库顶需要 Lead 新增
// `api.investigatorTools.commitDeckTop(seat)`（把揭示中的牌库顶牌从牌库移入当前技能检定的投入区，
// 复用 commit-rules 的合法性检查；不合法返回 false 且不移动）。
const deckTopRecord = (state, seat) => state?.scenarioFlags?.mechanics?.eote?.deckTop?.seats?.[seat] || null

export default {
  code: '08005',
  abilities: [
    {
      kind: 'free',
      cost: { exhaust: true },
      label: '交换牌库顶与一张手牌',
      criteria: ({ inv, state }) => Boolean(deckTopRecord(state, inv.seat)) && (inv.hand || []).length > 0,
      resolve: ({ inv, state, api }) => {
        const record = deckTopRecord(state, inv.seat)
        if (!record) return false
        const codes = [...new Set(inv.hand)]
        if (!codes.length) return false
        const options = codes.map(code => ({ id: code, label: api.cardName(code) }))
        const answer = options.length === 1 ? options[0].id
          : api.choice({ kind: 'effect', source: '08005', seat: inv.seat, prompt: '选择与牌库顶交换的手牌', options })
        if (!codes.includes(answer)) return false
        const movedDown = api.moveCard({ code: record.code, from: 'deck', to: 'hand' })
        const movedUp = movedDown && api.moveCard({ code: answer, from: 'hand', to: 'deckTop' })
        if (!movedUp) return false
        api.deckTop.clear(inv.seat)
        api.deckTop.reveal(inv.seat, { public: false, discount: 1 })
        return true
      },
    },
    {
      kind: 'free',
      cost: { exhaust: true },
      label: '把牌库顶的卡牌投入到当前技能检定',
      criteria: ({ inv, state }) => Boolean(state.skillTest) && Boolean(deckTopRecord(state, inv.seat)),
      resolve: ({ inv, api }) => {
        // P5 请求项：把揭示中的牌库顶牌投入当前检定（调用方在合法时移动并计入投入图标）。
        if (typeof api.investigatorTools?.commitDeckTop !== 'function') return false
        return api.investigatorTools.commitDeckTop(inv.seat)
      },
    },
  ],
}

```
